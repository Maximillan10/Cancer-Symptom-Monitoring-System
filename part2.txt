<?php

namespace App\Http\Controllers\Web;

use App\Http\Controllers\Controller;
use App\Models\Patient;
use App\Models\SymptomSubmission;
use App\Support\Severity;
use Illuminate\Http\Request;
use Illuminate\Support\Str;
use Illuminate\View\View;

class DashboardController extends Controller
{
    public function index(Request $request): View
    {
        $user = $request->user();
        $search = trim((string) $request->input('search', ''));

        // A clinician sees their own patients; an admin sees everyone.
        $all = Patient::query()
            ->when(! $user->isAdmin(), fn ($q) => $q->where('assigned_clinician_id', $user->id))
            ->with('latestSubmission')
            ->withCount('unresolvedAlerts')
            ->get();

        // Triage order: highest current symptom score first, then name.
        $rows = $all->map(function (Patient $patient) {
            $latest = $patient->latestSubmission;
            $top = Severity::top($latest);

            return [
                'patient' => $patient,
                'latest' => $latest,
                'top' => $top,
                'score' => $top['score'] ?? -1,
            ];
        })->sort(function ($a, $b) {
            return [$b['score'], $a['patient']->name] <=> [$a['score'], $b['patient']->name];
        })->values();

        $stats = [
            'patients' => $all->count(),
            'checkins' => SymptomSubmission::whereIn('patient_id', $all->pluck('id'))
                ->where('submitted_at', '>=', now()->subDays(7))
                ->count(),
            'high' => $rows->where('score', '>=', Severity::HIGH)->count(),
            'alerts' => (int) $all->sum('unresolved_alerts_count'),
        ];

        $visible = $search === ''
            ? $rows
            : $rows->filter(fn ($r) => Str::contains(Str::lower($r['patient']->name), Str::lower($search)))->values();

        return view('dashboard', [
            'rows' => $visible,
            'stats' => $stats,
            'search' => $search,
            'types' => SymptomSubmission::SYMPTOM_TYPES,
        ]);
    }
}
<?php

namespace App\Models;

use App\Services\AlertRules;
use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

class Alert extends Model
{
    use HasFactory;

    protected $fillable = [
        'patient_id',
        'triggered_at',
        'symptom_type',
        'severity',
        'threshold_breached',
        'resolved_at',
        'resolved_by',
    ];

    protected function casts(): array
    {
        return [
            'triggered_at' => 'datetime',
            'severity' => 'integer',
            'resolved_at' => 'datetime',
        ];
    }

    public function patient(): BelongsTo
    {
        return $this->belongsTo(Patient::class);
    }

    public function resolver(): BelongsTo
    {
        return $this->belongsTo(User::class, 'resolved_by');
    }

    /** Alerts nobody has resolved yet. */
    public function scopeOpen(Builder $query): Builder
    {
        return $query->whereNull('resolved_at');
    }

    public function scopeResolved(Builder $query): Builder
    {
        return $query->whereNotNull('resolved_at');
    }

    /** A clinician sees alerts for their own patients; an admin sees all of them. */
    public function scopeVisibleTo(Builder $query, User $user): Builder
    {
        if ($user->isAdmin()) {
            return $query;
        }

        return $query->whereHas('patient', fn (Builder $patients) => $patients->where('assigned_clinician_id', $user->id));
    }

    public function isResolved(): bool
    {
        return $this->resolved_at !== null;
    }

    public function resolve(User $clinician): void
    {
        $this->update([
            'resolved_at' => now(),
            'resolved_by' => $clinician->id,
        ]);
    }

    /** critical, warning or informational (see AlertRules::priority). */
    public function priority(): string
    {
        return AlertRules::priority((int) $this->severity);
    }

    public function priorityLabel(): string
    {
        return ucfirst($this->priority());
    }

    /** Bootstrap badge classes for the priority. */
    public function priorityBadge(): string
    {
        return match ($this->priority()) {
            'critical' => 'text-bg-danger',
            'warning' => 'text-bg-warning',
            default => 'text-bg-info',
        };
    }
}
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;

class Patient extends Model
{
    use HasFactory;

protected $fillable = [
    'assigned_clinician_id',
    'name',
    'dob',
    'diagnosis',
    'mobile_app_user_id',
    'sex',
    'cancer_type',
    'cancer_stage',
];

    protected function casts(): array
    {
        return [
            'dob' => 'date',
        ];
    }

    public function clinician(): BelongsTo
    {
        return $this->belongsTo(User::class, 'assigned_clinician_id');
    }

    public function symptomSubmissions(): HasMany
    {
        return $this->hasMany(SymptomSubmission::class);
    }

    public function alerts(): HasMany
    {
        return $this->hasMany(Alert::class);
    }

    public function unresolvedAlerts(): HasMany
    {
        return $this->alerts()->whereNull('resolved_at');
    }

    public function reports(): HasMany
    {
        return $this->hasMany(Report::class);
    }

    /** Most recent symptom submission, useful for dashboard patient list view */
    public function latestSubmission()
    {
        return $this->hasOne(SymptomSubmission::class)->latestOfMany('submitted_at');
    }
}
<?php

namespace App\Policies;

use App\Models\Patient;
use App\Models\User;

/**
 * A clinician can only access patients assigned to them.
 * Admins can access every patient.
 * Laravel discovers this policy automatically (App\Policies\PatientPolicy for App\Models\Patient).
 */
class PatientPolicy
{
    public function viewAny(User $user): bool
    {
        return true; // the controller scopes the list to the clinician's own patients
    }

    public function view(User $user, Patient $patient): bool
    {
        return $user->isAdmin() || (int) $patient->assigned_clinician_id === (int) $user->id;
    }

    public function create(User $user): bool
    {
        return true;
    }

    public function update(User $user, Patient $patient): bool
    {
        return $this->view($user, $patient);
    }

    public function delete(User $user, Patient $patient): bool
    {
        return $this->view($user, $patient);
    }
}
